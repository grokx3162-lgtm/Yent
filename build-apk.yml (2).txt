name: Generar APK de Luna AI

on:
  push:
    branches:
      - main
      - master
  pull_request:
    branches:
      - main
      - master
  workflow_dispatch:

jobs:
  build-apk:
    name: Generar Luna-AI.apk
    runs-on: ubuntu-latest
    permissions:
      contents: write

    steps:
      - name: Descargar código
        uses: actions/checkout@v4

      - name: Configurar JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '17'

      - name: Configurar Android SDK
        uses: android-actions/setup-android@v3

      - name: Instalar Android SDK 36
        run: |
          sdkmanager "platforms;android-36" "platforms;android-36.1" "build-tools;36.0.0" || sdkmanager "platforms;android-36" "build-tools;36.0.0"

      - name: Configurar Gradle 9.3.1
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '9.3.1'

      - name: Decodificar debug.keystore (para Firebase y Google Sign-In)
        run: |
          if [ ! -f debug.keystore ] && [ -f debug.keystore.base64 ]; then
            base64 -d debug.keystore.base64 > debug.keystore
          fi

      - name: Preparar .env
        run: |
          if [ ! -f .env ] && [ -f .env.example ]; then
            cp .env.example .env
          fi

      - name: Compilar APK
        run: gradle :app:assembleDebug --no-daemon --stacktrace

      - name: Renombrar APK a Luna-AI.apk
        run: |
          APK_PATH=$(find app/build/outputs/apk/debug -name "*.apk" | head -n 1)
          cp "$APK_PATH" Luna-AI.apk

      - name: Subir APK a Releases (descarga directa sin ZIP)
        if: github.event_name != 'pull_request'
        uses: softprops/action-gh-release@v2
        with:
          tag_name: latest
          name: Luna AI - APK Listo para Instalar
          body: Descarga directa de `Luna-AI.apk` compilado automáticamente desde el commit `${{ github.sha }}`.
          files: Luna-AI.apk
          make_latest: true

      - name: Subir APK en Artifacts
        uses: actions/upload-artifact@v4
        with:
          name: Luna-AI-APK
          path: Luna-AI.apk
          if-no-files-found: error
          retention-days: 30
